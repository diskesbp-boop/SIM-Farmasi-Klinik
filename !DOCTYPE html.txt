                                                    link.click();
                                                    link.remove();
                                                    showAlert('Rencana pembelian & analisis Pareto ABC berhasil diunduh dalam format Excel/CSV!');
                                                }}
                                                className="bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-semibold px-3.5 py-2 rounded-lg shadow transition flex items-center gap-1.5"
                                            >
                                                <i className="fa-solid fa-file-excel"></i> Download Excel
                                            </button>
                                        </div>
                                    </div>

                                    <div className="overflow-x-auto">
                                        <table className="w-full text-left text-xs text-slate-600">
                                            <thead className="bg-slate-50 font-semibold uppercase text-slate-500 border-b">
                                                <tr>
                                                    <th className="p-3">Rank</th>
                                                    <th className="p-3">Nama Produk</th>
                                                    <th className="p-3 text-center">Kelas Pareto</th>
                                                    <th className="p-3 text-right">Total Keluar (30h)</th>
                                                    <th className="p-3 text-right">Konsumsi/Hari</th>
                                                    <th className="p-3 text-right">Safety Stock</th>
                                                    <th className="p-3 text-right">ROP</th>
                                                    <th className="p-3 text-right">Stok Saat Ini</th>
                                                    <th className="p-3 text-right font-bold text-blue-600">Rekomendasi Beli</th>
                                                </tr>
                                            </thead>
                                            <tbody className="divide-y">
                                                {purchasePlanList.map(row => (
                                                    <tr key={row.id} className="hover:bg-slate-50">
                                                        <td className="p-3 font-bold">{row.rank}</td>
                                                        <td className="p-3 font-medium text-slate-800">
                                                            <div>{row.name}</div>
                                                            <div className="text-[10px] text-slate-400 font-mono">{row.id}</div>
                                                        </td>
                                                        <td className="p-3 text-center">
                                                            <span className={`px-2.5 py-1 rounded-full text-xs font-bold ${
                                                                row.abcClass === 'A' ? 'bg-rose-100 text-rose-700' :
                                                                row.abcClass === 'B' ? 'bg-amber-100 text-amber-700' : 'bg-blue-100 text-blue-700'
                                                            }`}>
                                                                Kelas {row.abcClass}
                                                            </span>
                                                        </td>
                                                        <td className="p-3 text-right font-semibold">{row.totalOut} {row.unit}</td>
                                                        <td className="p-3 text-right">{row.consumptionRate}</td>
                                                        <td className="p-3 text-right">{row.safetyStock}</td>
                                                        <td className="p-3 text-right font-semibold">{row.rop}</td>
                                                        <td className="p-3 text-right font-bold">{row.stock}</td>
                                                        <td className="p-3 text-right font-extrabold text-blue-600 text-sm">
                                                            {row.suggestedOrder} {row.unit}
                                                        </td>

                                                ))}
                                            </tbody>
                                        </table>
                                    </div>
                                </div>
                            )}

                            {/* TAB 5: FAKTUR PEMBELIAN */}
                            {activeTab === 'incoming' && (
                                <div className="bg-white rounded-xl border border-slate-200 shadow-sm p-6 space-y-6">
                                    <div className="border-b pb-4">
                                        <h2 className="text-lg font-bold text-slate-800">Faktur Pembelian (Penerimaan PBF / Diskes)</h2>
                                        <p className="text-sm text-slate-500">Input faktur resmi pengadaan obat lengkap dengan kalkulasi diskon, pajak, batch, dan jatuh tempo.</p>
                                    </div>

                                    <form onSubmit={handleSaveInvoice} className="space-y-6">
                                        <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 text-xs">
                                            <div>
                                                <label className="block font-semibold text-slate-600 mb-1">Supplier *</label>
                                                <select value={invoiceHeader.supplier} onChange={e => setInvoiceHeader({...invoiceHeader, supplier: e.target.value})} className="w-full border rounded-lg p-2 text-xs">
                                                    <option value="Diskes Kodaeral XIV">Diskes Kodaeral XIV</option>
                                                    <option value="PT. Kimia Farma">PT. Kimia Farma</option>
                                                    <option value="PT. Kalbe Farma">PT. Kalbe Farma</option>

                                                <label className="block font-semibold text-slate-600 mb-1">No. Surat Pesanan *</label>

                                            </div>
                                            <div>
                                                <label className="block font-semibold text-slate-600 mb-1">Tanggal Faktur</label>
                                                <input type="date" value={invoiceHeader.invoiceDate} onChange={e => setInvoiceHeader({...invoiceHeader, invoiceDate: e.target.value})} className="w-full border rounded-lg p-2 text-xs" />
                                            </div>
                                            <div>
                                                <label className="block font-semibold text-slate-600 mb-1">Tanggal Penerimaan</label>
                                                <input type="text" value={invoiceHeader.receiveDate} onChange={e => setInvoiceHeader({...invoiceHeader, receiveDate: e.target.value})} className="w-full border rounded-lg p-2 text-xs" />
                                            </div>
                                        </div>

                                        <div className="overflow-x-auto border rounded-xl">
                                            <table className="w-full text-left text-xs text-slate-600">
                                                <thead className="bg-slate-50 font-semibold border-b text-slate-600">
                                                    <tr>
                                                        <th className="p-2.5 w-10 text-center">No.</th>
                                                        <th className="p-2.5">Produk</th>
                                                        <th className="p-2.5 w-32">Expired Date</th>
                                                        <th className="p-2.5 w-32">No. Batch</th>
                                                        <th className="p-2.5 w-24">Kuantitas</th>
                                                        <th className="p-2.5 w-24">Satuan</th>
                                                        <th className="p-2.5 w-28">Harga Beli</th>
                                                        <th className="p-2.5 w-24">Diskon (Rp)</th>
                                                        <th className="p-2.5 w-28">Sub Total</th>
                                                        <th className="p-2.5 w-10"></th>
                                                    </tr>
                                                </thead>
                                                <tbody className="divide-y">
                                                    {invoiceItems.map((row, idx) => {
                                                        const lineSub = (Number(row.qty) || 0) * (Number(row.buyPrice) || 0) - (Number(row.discount) || 0);
                                                        return (
                                                            <tr key={idx}>
                                                                <td className="p-2 text-center">{idx + 1}</td>
                                                                <td className="p-2">
                                                                    <select value={row.itemId} onChange={e => handleInvoiceItemChange(idx, 'itemId', e.target.value)} className="w-full border rounded p-1.5 text-xs">
                                                                        <option value="">Pilih Produk</option>
                                                                        {items.map(i => <option key={i.id} value={i.id}>{i.name}</option>)}
                                                                    </select>
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="date" value={row.expDate} onChange={e => handleInvoiceItemChange(idx, 'expDate', e.target.value)} className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="text" value={row.batchNo} onChange={e => handleInvoiceItemChange(idx, 'batchNo', e.target.value)} placeholder="No Batch" className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="number" min="1" value={row.qty} onChange={e => handleInvoiceItemChange(idx, 'qty', e.target.value)} className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="text" value={row.unit} onChange={e => handleInvoiceItemChange(idx, 'unit', e.target.value)} className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="number" value={row.buyPrice} onChange={e => handleInvoiceItemChange(idx, 'buyPrice', e.target.value)} className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2">
                                                                    <input type="number" value={row.discount} onChange={e => handleInvoiceItemChange(idx, 'discount', e.target.value)} className="w-full border rounded p-1.5 text-xs" />
                                                                </td>
                                                                <td className="p-2 font-bold text-slate-800">
                                                                    Rp {lineSub.toLocaleString()}
                                                                </td>
                                                                <td className="p-2 text-center">
                                                                    {invoiceItems.length > 1 && (
                                                                        <button type="button" onClick={() => handleRemoveInvoiceItem(idx)} className="text-rose-500 hover:text-rose-700">
                                                                            <i className="fa-solid fa-trash"></i>
                                                                        </button>
                                                                    )}
                                                                </td>
                                                            </tr>
                                                        );
                                                    })}
                                                </tbody>
                                            </table>
                                        </div>

                                        <button type="button" onClick={handleAddInvoiceItem} className="text-xs text-blue-600 font-semibold hover:underline flex items-center gap-1">
                                            <i className="fa-solid fa-bolt"></i> Tambah Produk Sekaligus / Baris Baru
                                        </button>

                                        <div className="flex flex-col md:flex-row justify-between items-start gap-6 border-t pt-4 text-xs">
                                            <div></div>
                                            <div className="w-full md:w-80 space-y-2">
                                                <div className="flex justify-between items-center">
                                                    <span className="text-slate-500">Sub Total:</span>
                                                    <span className="font-bold">Rp {calculateSubTotal().toLocaleString()}</span>
                                                </div>
                                                <div className="flex justify-between items-center gap-2">
                                                    <span className="text-slate-500">Diskon Faktur (%):</span>
                                                    <input type="number" value={invoiceHeader.discountPercent} onChange={e => setInvoiceHeader({...invoiceHeader, discountPercent: e.target.value})} className="w-24 border rounded p-1 text-right text-xs" />
                                                </div>
                                                <div className="flex justify-between items-center gap-2">
                                                    <span className="text-slate-500">Biaya Lainnya:</span>
                                                    <input type="number" value={invoiceHeader.otherCosts} onChange={e => setInvoiceHeader({...invoiceHeader, otherCosts: e.target.value})} className="w-24 border rounded p-1 text-right text-xs" />
                                                </div>
                                                <div className="flex justify-between items-center text-sm border-t pt-2">
                                                    <span className="font-bold text-slate-800">TOTAL:</span>
                                                    <span className="font-extrabold text-blue-600 text-base">Rp {calculateGrandTotal().toLocaleString()}</span>
                                                </div>
                                                <button type="submit" className="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-bold py-2.5 rounded-lg shadow-md transition mt-3">
                                                    Simpan Faktur Pembelian
                                                </button>
                                            </div>
                                        </div>
                                    </form>
                                </div>
                            )}

                            {/* TAB: RETUR PEMBELIAN (BERDASARKAN FAKTUR PEMBELIAN PBF/DISKES) */}
                            {activeTab === 'returns' && (() => {
                                const incomingTrxs = transactions.filter(t => t.type === 'IN');
                                const invoiceList = Array.from(new Set(incomingTrxs.map(t => t.docNo)))
                                    .map(docNo => {
                                        const sample = incomingTrxs.find(t => t.docNo === docNo);
                                        return {
                                            invoiceNo: docNo,
                                            supplier: sample?.notes?.replace('Faktur PBF: ', '') || 'PBF / Diskes',
                                            date: sample?.date || ''
                                        };
                                    });

                                const handleInvoiceSelect = (e) => {
                                    const invNo = e.target.value;
                                    const targetInv = invoiceList.find(i => i.invoiceNo === invNo);
                                    setPurchaseReturnForm({
                                        invoiceNo: invNo,
                                        supplier: targetInv ? targetInv.supplier : '',
                                        returnDate: purchaseReturnForm.returnDate,
                                        items: [{ itemId: '', qty: '', unit: '', reason: 'Mendekati Kadaluwarsa (ED)' }]
                                    });
                                };

                                const handleAddItemRow = () => {
                                    setPurchaseReturnForm(prev => ({
                                        ...prev,
                                        items: [...(prev.items || []), { itemId: '', qty: '', unit: '', reason: 'Mendekati Kadaluwarsa (ED)' }]
                                    }));
                                };

                                const handleRemoveItemRow = (index) => {
                                    setPurchaseReturnForm(prev => {
                                        const list = prev.items || [];
                                        if (list.length === 1) return prev;
                                        return {
                                            ...prev,
                                            items: list.filter((_, idx) => idx !== index)
                                        };
                                    });
                                };

                                const handleItemRowChange = (index, field, value) => {
                                    setPurchaseReturnForm(prev => {
                                        const updatedItems = [...(prev.items || [])];
                                        if (!updatedItems[index]) {
                                            updatedItems[index] = { itemId: '', qty: '', unit: '', reason: 'Mendekati Kadaluwarsa (ED)' };
                                        }
                                        updatedItems[index][field] = value;
                                        if (field === 'itemId') {
                                            const itemObj = items.find(i => i.id === value);
                                            updatedItems[index].unit = itemObj ? itemObj.unit : '';
                                        }
                                        return {
                                            ...prev,
                                            items: updatedItems
                                        };
                                    });
                                };

                                const handlePurchaseReturnSubmit = (e) => {
                                    e.preventDefault();
                                    if (!purchaseReturnForm.invoiceNo) {
                                        showAlert('Mohon pilih Nomor Faktur Pembelian!', 'error');
                                        return;
                                    }
                                    const formItems = purchaseReturnForm.items || [];
                                    const validRows = formItems.filter(i => i && i.itemId && Number(i.qty) > 0);
                                    if (validRows.length === 0) {
                                        showAlert('Mohon lengkapi minimal 1 baris barang retur dengan kuantitas valid!', 'error');
                                        return;
                                    }

                                    let insufficient = false;
                                    validRows.forEach(row => {
                                        const targetItem = items.find(i => i.id === row.itemId);
                                        if (targetItem && targetItem.stock < Number(row.qty)) {
                                            insufficient = true;
                                            showAlert(`Stok fisik ${targetItem.name} tidak mencukupi (Sisa: ${targetItem.stock}) untuk diretur!`, 'error');
                                        }
                                    });
                                    if (insufficient) return;

                                    let updatedItems = [...items];
                                    let newTrxs = [];

                                    validRows.forEach(row => {
                                        const qtyVal = parseInt(row.qty, 10);
                                        const targetItem = updatedItems.find(i => i.id === row.itemId);
                                        
                                        updatedItems = updatedItems.map(i => {
                                            if (i.id === row.itemId) {
                                                return { ...i, stock: i.stock - qtyVal };
                                            }
                                            return i;
                                        });

                                        newTrxs.push({
                                            id: 'TRX-RTP-' + Date.now().toString().slice(-5) + '-' + Math.floor(Math.random()*10),
                                            date: purchaseReturnForm.returnDate,
                                            itemId: row.itemId,
                                            itemName: targetItem ? targetItem.name : '',
                                            type: 'OUT',
                                            qty: qtyVal,
                                            docNo: `RET-FAK/${purchaseReturnForm.invoiceNo}`,
                                            notes: `Retur Pembelian [Ke: ${purchaseReturnForm.supplier}] - Ref Faktur: ${purchaseReturnForm.invoiceNo} - Alasan: ${row.reason}`
                                        });
                                    });

                                    setItems(updatedItems);
                                    setTransactions([...newTrxs, ...transactions]);
                                    showAlert(`Retur pembelian untuk ${validRows.length} item barang ke PBF/Diskes berhasil diproses & stok dipotong!`);
                                    setPurchaseReturnForm({
                                        invoiceNo: '',
                                        supplier: '',
                                        returnDate: new Date().toISOString().split('T')[0],
                                        items: [{ itemId: '', qty: '', unit: '', reason: 'Mendekati Kadaluwarsa (ED)' }]
                                    });
                                };

                                const purchaseReturnLogs = transactions.filter(t => t.notes && t.notes.includes('Retur Pembelian'));
                                const currentReturnItems = purchaseReturnForm.items || [{ itemId: '', qty: '', unit: '', reason: 'Mendekati Kadaluwarsa (ED)' }];

                                return (
                                    <div className="bg-white rounded-xl border border-slate-200 shadow-sm p-6 space-y-6">
                                        <div className="border-b pb-4 flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
                                            <div>
                                                <h2 className="text-lg font-bold text-slate-800">Retur Pembelian Sediaan (Ke Supplier / Diskes)</h2>
                                                <p className="text-sm text-slate-500">Pengembalian obat/BMHP ke PBF atau Dinas Kesehatan berdasarkan Faktur Pembelian yang telah diterima.</p>
                                            </div>
                                            <button 
                                                onClick={() => {
                                                    const headers = ["ID Transaksi", "Tanggal Retur", "No. Faktur Pembelian", "Supplier / PBF", "Nama Barang", "Kuantitas Retur", "Alasan Retur"];
                                                    const rows = purchaseReturnLogs.map(log => [
                                                        log.id,
                                                        log.date,
                                                        log.docNo.replace('RET-FAK/', ''),
                                                        `"${log.notes.split(' - ')[0].replace('Retur Pembelian [Ke: ', '').replace(']', '')}"`,
                                                        `"${log.itemName}"`,
                                                        log.qty,
                                                        `"${log.notes}"`
                                                    ]);
                                                    const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
                                                    const encodedUri = encodeURI(csvContent);
                                                    const link = document.createElement("a");
                                                    link.setAttribute("href", encodedUri);
                                                    link.setAttribute("download", `Retur_Pembelian_PBF_${new Date().toISOString().slice(0,10)}.csv`);
                                                    document.body.appendChild(link);
                                                    link.click();
                                                    link.remove();
                                                    showAlert('Riwayat retur pembelian berhasil diunduh dalam format Excel/CSV!');
                                                }}
                                                className="bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-semibold px-3.5 py-2 rounded-lg shadow transition flex items-center gap-1.5"
                                            >
                                                <i className="fa-solid fa-file-excel"></i> Download Excel
                                            </button>
                                        </div>

                                        <form onSubmit={handlePurchaseReturnSubmit} className="bg-slate-50 p-4 rounded-xl border border-slate-200 space-y-4">
                                            <h3 className="text-sm font-bold text-slate-800 flex items-center gap-2">
                                                <i className="fa-solid fa-file-invoice-dollar text-blue-600"></i> Input Form Retur Multi-Item Berdasarkan Faktur Pembelian
                                            </h3>

                                            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 text-xs">
                                                <div>
                                                    <label className="block font-semibold text-slate-600 mb-1">Pilih No. Faktur Pembelian *</label>
                                                    <select value={purchaseReturnForm.invoiceNo} onChange={handleInvoiceSelect} className="w-full border rounded-lg p-2.5 text-xs bg-white outline-none focus:ring-2 focus:ring-blue-500 font-mono" required>
                                                        <option value="">-- Pilih Faktur Pembelian --</option>
                                                        {invoiceList.map(inv => (
                                                            <option key={inv.invoiceNo} value={inv.invoiceNo}>
                                                                Faktur: {inv.invoiceNo} ({inv.supplier} - {inv.date})
                                                            </option>
                                                        ))}
                                                    </select>
                                                </div>

                                                <div>
                                                    <label className="block font-semibold text-slate-600 mb-1">Supplier / Pengirim</label>
                                                    <input type="text" value={purchaseReturnForm.supplier} placeholder="Otomatis dari faktur" readOnly className="w-full border rounded-lg p-2.5 text-xs bg-slate-100 text-slate-600 font-semibold cursor-not-allowed" />
                                                </div>

                                                <div>
                                                    <label className="block font-semibold text-slate-600 mb-1">Tanggal Retur *</label>
                                                    <input type="date" value={purchaseReturnForm.returnDate} onChange={e => setPurchaseReturnForm({...purchaseReturnForm, returnDate: e.target.value})} className="w-full border rounded-lg p-2.5 text-xs bg-white outline-none focus:ring-2 focus:ring-blue-500" required />
                                                </div>
                                            </div>

                                            {/* MULTI-ITEM RETUR TABLE */}
                                            <div className="space-y-3 pt-2">
                                                <label className="block text-xs font-semibold text-slate-700 uppercase">Daftar Sediaan Barang yang Diretur *</label>
                                                {currentReturnItems.map((row, idx) => (
                                                    <div key={idx} className="flex flex-col sm:flex-row items-center gap-3 bg-white p-3 rounded-lg border border-slate-200 shadow-sm">
                                                        <div className="flex-1 w-full">
                                                            <label className="block text-[10px] text-slate-400 font-semibold mb-0.5">Nama Barang</label>
                                                            <select 
                                                                value={row?.itemId || ''} 
                                                                onChange={e => handleItemRowChange(idx, 'itemId', e.target.value)} 
                                                                className="w-full border rounded-lg p-2 text-xs bg-white outline-none focus:ring-2 focus:ring-blue-500" 
                                                                required
                                                            >
                                                                <option value="">-- Pilih Barang --</option>
                                                                {items.map(i => (
                                                                    <option key={i.id} value={i.id}>{i.name} (Stok: {i.stock} {i.unit})</option>

                                                            </select>
                                                        </div>

                                                        <div className="w-24">
                                                            <label className="block text-[10px] text-slate-400 font-semibold mb-0.5">Kuantitas</label>
                                                            <input 
                                                                type="number" 
                                                                min="1" 
                                                                value={row?.qty || ''} 
                                                                onChange={e => handleItemRowChange(idx, 'qty', e.target.value)} 
                                                                placeholder="Qty" 
                                                                className="w-full border rounded-lg p-2 text-xs text-center" 
                                                                required 
                                                            />
                                                        </div>

                                                        <div className="w-24">
                                                            <label className="block text-[10px] text-slate-400 font-semibold mb-0.5">Satuan</label>
                                                            <input 
                                                                type="text" 
                                                                value={row?.unit || ''} 
                                                                placeholder="Satuan" 
                                                                readOnly 
                                                                className="w-full border rounded-lg p-2 text-xs bg-slate-100 text-center font-semibold cursor-not-allowed" 
                                                           />
                                                        </div>

                                                        <div className="flex-1 w-full">
                                                            <label className="block text-[10px] text-slate-400 font-semibold mb-0.5">Alasan Retur</label>
                                                            <select 
                                                                value={row?.reason || 'Mendekati Kadaluwarsa (ED)'} 
                                                                onChange={e => handleItemRowChange(idx, 'reason', e.target.value)} 
                                                                className="w-full border rounded-lg p-2 text-xs bg-white outline-none"
                                                            >
                                                                <option value="Mendekati Kadaluwarsa (ED)">Mendekati Kadaluwarsa (ED)</option>
                                                                <option value="Kemasan Rusak / Cacat Fisik">Kemasan Rusak / Cacat Fisik</option>
                                                                <option value="Tidak Sesuai Surat Pesanan (SP)">Tidak Sesuai Surat Pesanan (SP)</option>
                                                                <option value="Produk Terbit Recall BPOM">Produk Terbit Recall BPOM</option>
                                                            </select>
                                                        </div>

                                                        {currentReturnItems.length > 1 && (
                                                            <button 
                                                                type="button" 
                                                                onClick={() => handleRemoveItemRow(idx)} 
                                                                className="text-rose-500 hover:text-rose-700 p-2 self-end sm:self-center"
                                                            >
                                                                <i className="fa-solid fa-trash text-sm"></i>
                                                            </button>
                                                        )}
                                                    </div>
                                                ))}

                                                <button 
                                                    type="button" 
                                                    onClick={handleAddItemRow} 
                                                    className="text-xs text-blue-600 font-semibold hover:underline flex items-center gap-1 mt-1"
                                                >
                                                    <i className="fa-solid fa-plus"></i> Tambah Baris Barang Lainnya
                                                </button>
                                            </div>

                                            <div className="pt-2 flex justify-end">
                                                <button type="submit" className="bg-rose-600 hover:bg-rose-700 text-white text-xs font-bold px-5 py-2.5 rounded-lg shadow transition flex items-center gap-2">
                                                    <i className="fa-solid fa-rotate-left"></i> Proses &amp; Kurangi Stok Retur Pembelian
                                                </button>
                                            </div>
                                        </form>

                                        <div>
                                            <h3 className="text-sm font-bold text-slate-800 mb-3">Riwayat Transaksi Retur Pembelian ke PBF/Diskes</h3>
                                            <div className="overflow-x-auto border rounded-xl">
                                                <table className="w-full text-left text-xs text-slate-600">
                                                    <thead className="bg-slate-50 font-semibold uppercase text-slate-500 border-b">
                                                        <tr>
                                                            <th className="p-3">Tanggal Retur</th>
                                                            <th className="p-3">ID Transaksi</th>
                                                            <th className="p-3">Ref. No Faktur</th>
                                                            <th className="p-3">Nama Barang</th>
                                                            <th className="p-3 text-right">Kuantitas Retur</th>
                                                            <th className="p-3">Detail &amp; Alasan Retur</th>
                                                        </tr>
                                                    </thead>
                                                    <tbody className="divide-y">
                                                        {purchaseReturnLogs.length === 0 ? (
                                                            <tr>
                                                                <td colSpan="6" className="p-6 text-center text-slate-400 italic">Belum ada riwayat retur pembelian berdasarkan faktur.</td>
                                                            </tr>
                                                        ) : (
                                                            purchaseReturnLogs.map((log, idx) => (
                                                                <tr key={idx} className="hover:bg-slate-50">
                                                                    <td className="p-3">{log.date}</td>
                                                                    <td className="p-3 font-mono text-xs">{log.id}</td>
                                                                    <td className="p-3 font-mono text-xs font-bold text-blue-600">{log.docNo.replace('RET-FAK/', '')}</td>
                                                                    <td className="p-3 font-medium text-slate-800">{log.itemName}</td>
                                                                    <td className="p-3 text-right font-bold text-rose-600 text-sm">-{log.qty}</td>
                                                                    <td className="p-3 text-xs text-slate-600">{log.notes}</td>
                                                                </tr>
                                                            ))
                                                        )}
                                                    </tbody>
                                                </table>
                                            </div>
                                        </div>
                                    </div>
                                );
                            })()}

                            {/* TAB: STOK OPNAME */}
                            {activeTab === 'stock_opname' && (() => {
                                const handlePhysicalChange = (itemId, val) => {
                                    setPhysicalCounts(prev => ({ ...prev, [itemId]: val }));
                                };

                                const handleReasonChange = (itemId, val) => {
                                    setOpnameReasons(prev => ({ ...prev, [itemId]: val }));
                                };

                                const handleSaveStockOpname = (e) => {
                                    e.preventDefault();
                                    let updatedItems = [...items];
                                    let newTrxs = [];
                                    let adjustedCount = 0;

                                    items.forEach(item => {
                                        const physVal = physicalCounts[item.id];
                                        if (physVal !== undefined && physVal !== '' && !isNaN(physVal)) {
                                            const physicalQty = parseInt(physVal, 10);
                                            const diff = physicalQty - item.stock;

                                            if (diff !== 0) {
                                                adjustedCount++;
                                                updatedItems = updatedItems.map(i => {
                                                    if (i.id === item.id) {
                                                        return { ...i, stock: physicalQty };
                                                    }
                                                    return i;
                                                });

                                                newTrxs.push({
                                                    id: 'TRX-SO-' + Date.now().toString().slice(-5) + '-' + Math.floor(Math.random()*10),
                                                    date: opnameDate,
                                                    itemId: item.id,
                                                    itemName: item.name,
                                                    type: diff > 0 ? 'IN' : 'OUT',
                                                    qty: Math.abs(diff),
                                                    docNo: `SO/${opnameDate.replace(/-/g,'')}`,
                                                    notes: `Stok Opname [Sistem: ${item.stock} -> Fisik: ${physicalQty}] (Selisih: ${diff > 0 ? '+'+diff : diff}) - Ket: ${opnameReasons[item.id] || opnameNotes || 'Audit Berkala'}`
                                                });
                                            }
                                        }
                                    });

                                    setItems(updatedItems);
                                    setTransactions([...newTrxs, ...transactions]);
                                    showAlert(`Stok Opname berhasil disimpan! ${adjustedCount} item sediaan disesuaikan dengan stok fisik.`);
                                };

                                return (
                                    <div className="bg-white rounded-xl border border-slate-200 shadow-sm p-6 space-y-6">
                                        <div className="border-b pb-4 flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
                                            <div>
                                                <h2 className="text-lg font-bold text-slate-800">Stok Opname Sediaan Farmasi &amp; BMHP</h2>
                                                <p className="text-sm text-slate-500">Pencocokan stok sistem dengan hasil perhitungan fisik di gudang / instalasi farmasi.</p>
                                            </div>
                                            <div className="flex flex-wrap items-center gap-2">
                                                <button 
                                                    onClick={() => {
                                                        const headers = ["ID Produk", "Nama Produk", "Kategori", "Stok Sistem", "Satuan", "No Batch", "Exp Date"];
                                                        const rows = items.map(i => [i.id, `"${i.name}"`, i.category, i.stock, i.unit, i.batchNo, i.expDate]);
                                                        const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
                                                        const encodedUri = encodeURI(csvContent);
                                                        const link = document.createElement("a");
                                                        link.setAttribute("href", encodedUri);
                                                        link.setAttribute("download", `Lembar_Kerja_Stok_Opname_${new Date().toISOString().slice(0,10)}.csv`);
                                                        document.body.appendChild(link);
                                                        link.click();
                                                        link.remove();
                                                        showAlert('Lembar kerja Stok Opname berhasil diunduh!');
                                                    }}
                                                    className="bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-semibold px-3.5 py-2 rounded-lg shadow transition flex items-center gap-1.5"
                                                >
                                                    <i className="fa-solid fa-file-excel"></i> Download Lembar Kerja Excel
                                                </button>
                                            </div>
                                        </div>

                                        <form onSubmit={handleSaveStockOpname} className="space-y-4">
                                            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 bg-slate-50 p-4 rounded-xl border border-slate-200 text-xs">
                                                <div>
                                                    <label className="block font-semibold text-slate-600 mb-1">Tanggal Pelaksanaan Stok Opname *</label>
                                                    <input type="date" value={opnameDate} onChange={e => setOpnameDate(e.target.value)} className="w-full border rounded-lg p-2 bg-white outline-none focus:ring-2 focus:ring-blue-500 text-xs" required />
                                                </div>
                                                <div>
                                                    <label className="block font-semibold text-slate-600 mb-1">Keterangan / Berita Acara Umum</label>
                                                    <input type="text" value={opnameNotes} onChange={e => setOpnameNotes(e.target.value)} placeholder="Contoh: Audit Rutin Bulanan Instalasi Farmasi" className="w-full border rounded-lg p-2 bg-white outline-none focus:ring-2 focus:ring-blue-500 text-xs" />
                                                </div>
                                            </div>

                                            <div className="overflow-x-auto border rounded-xl">
                                                <table className="w-full text-left text-xs text-slate-600">
                                                    <thead className="bg-slate-50 font-semibold uppercase text-slate-500 border-b">
                                                        <tr>
                                                            <th className="p-3">Kode</th>
                                                            <th className="p-3">Nama Produk</th>
                                                            <th className="p-3">Kategori</th>
                                                            <th className="p-3 text-center">Stok Sistem</th>
                                                            <th className="p-3 text-center w-36">Stok Fisik (Input)</th>
                                                            <th className="p-3 text-center">Selisih</th>
                                                            <th className="p-3">Catatan / Alasan Selisih</th>
                                                        </tr>
                                                    </thead>
                                                    <tbody className="divide-y">
                                                        {items.map(item => {
                                                            const physicalVal = physicalCounts[item.id] !== undefined ? physicalCounts[item.id] : item.stock;
                                                            const diff = (parseInt(physicalVal, 10) || 0) - item.stock;
                                                            return (
                                                                <tr key={item.id} className="hover:bg-slate-50">
                                                                    <td className="p-3 font-mono text-xs">{item.id}</td>
                                                                    <td className="p-3 font-medium text-slate-800">
                                                                        <div>{item.name}</div>
                                                                        <div className="text-[10px] text-slate-400">Batch: {item.batchNo} | Exp: {item.expDate}</div>
                                                                    </td>
                                                                    <td className="p-3">{item.category}</td>
                                                                    <td className="p-3 text-center font-bold text-slate-800">{item.stock} {item.unit}</td>
                                                                    <td className="p-3 text-center">
                                                                        <input 
                                                                            type="number" 
                                                                            min="0" 
                                                                            value={physicalVal} 
                                                                            onChange={e => handlePhysicalChange(item.id, e.target.value)} 
                                                                            className="w-28 border rounded-lg p-2 text-center font-bold text-blue-600 text-xs bg-white focus:ring-2 focus:ring-blue-500 outline-none" 
                                                                            required 
                                                                        />
                                                                    </td>
                                                                    <td className={`p-3 text-center font-extrabold ${diff > 0 ? 'text-emerald-600' : diff < 0 ? 'text-rose-600' : 'text-slate-400'}`}>
                                                                        {diff > 0 ? `+${diff}` : diff}
                                                                    </td>
                                                                    <td className="p-3">
                                                                        <input 
                                                                            type="text" 
                                                                            value={opnameReasons[item.id] || ''} 
                                                                            onChange={e => handleReasonChange(item.id, e.target.value)} 
                                                                            placeholder="Sesuai / Rusak / Hilang..." 
                                                                            className="w-full border rounded p-1.5 text-xs bg-white outline-none" 
                                                                        />
                                                                    </td>
                                                                </tr>
                                                            );
                                                        })}
                                                    </tbody>
                                                </table>
                                            </div>

                                            <div className="pt-2 flex justify-end">
                                                <button type="submit" className="bg-blue-600 hover:bg-blue-700 text-white text-xs font-bold px-6 py-3 rounded-lg shadow-md transition flex items-center gap-2">
                                                    <i className="fa-solid fa-clipboard-check"></i> Simpan &amp; Sesuaikan Stok Opname Samaan
                                                </button>
                                            </div>
                                        </form>
                                    </div>
                                );
                            })()}

                            {/* OTHER PLACEHOLDER TABS */}
                            {['adjustment'].includes(activeTab) && (
                                <div className="bg-white rounded-xl border border-slate-200 shadow-sm p-6 text-center space-y-3">
                                    <div className="text-4xl text-blue-500"><i className="fa-solid fa-list-check"></i></div>
                                    <h2 className="text-base font-bold text-slate-800 capitalize">Modul {activeTab.replace('_', ' ')}</h2>
                                    <p className="text-xs text-slate-500 max-w-md mx-auto">Modul ini telah terintegrasi dengan database stok farmasi utama dan siap digunakan untuk proses audit LPLPO Klinik.</p>
                                </div>
                            )}
                        </main>
                    </div>
                </div>
            );
        }

        const container = document.getElementById('root');
        const root = ReactDOM.createRoot(container);
        root.render(<App />);
    </script>
</body>
</html>
