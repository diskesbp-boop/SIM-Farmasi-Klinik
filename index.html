// IMPORT EXCEL (.xlsx / .xls) DENGAN KONVERSI TANGGAL & SATUAN OTOMATIS const handleImportProductsExcel = (e) => { const file = e.target.files && e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = (evt) => { try { const data = new Uint8Array(evt.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheetName = workbook.SheetNames[0]; const worksheet = workbook.Sheets[firstSheetName]; const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1 }); if (jsonData.length < 2) { showAlert('File Excel kosong atau format tidak valid.', 'error'); return; } const headers = jsonData[0].map(h => String(h).trim().toLowerCase()); const getIdx = (keywords) => { for (let kw of keywords) { const idx = headers.findIndex(h => h.includes(kw)); if (idx >= 0) return idx; } return -1; }; const idxId = getIdx(['kode', 'id']); const idxName = getIdx(['nama produk', 'nama', 'name']); const idxCat = getIdx(['kategori', 'category']); const idxStock = getIdx(['stok', 'stock']); const idxUnit = getIdx(['satuan', 'unit']); const idxBatch = getIdx(['batch', 'no batch']); const idxExp = getIdx(['exp', 'expired', 'date']); const idxPrice = getIdx(['harga', 'price']); const idxMin = getIdx(['min', 'minimum']); // Helper untuk konversi tanggal Excel (Serial number -> YYYY-MM-DD) const parseExcelDate = (val) => { if (!val) return '2027-12-31'; if (typeof val === 'number') { const d = new Date(1899, 12, 30); d.setDate(d.getDate() + val); return !isNaN(d.getTime()) ? d.toISOString().split('T')[0] : '2027-12-31'; } const str = String(val).trim(); if (/^\d{5}$/.test(str)) { const num = Number(str); const d = new Date(1899, 12, 30); d.setDate(d.getDate() + num); return !isNaN(d.getTime()) ? d.toISOString().split('T')[0] : '2027-12-31'; } return str; }; let newItems = [...items]; let importedCount = 0; for (let i = 1; i < jsonData.length; i++) { const row = jsonData[i]; if (!row || row.length === 0) continue; const id = idxId >= 0 && row[idxId] !== undefined ? row[idxId] : ('OBT-' + Date.now() + '-' + i); const name = idxName >= 0 && row[idxName] !== undefined ? row[idxName] : 'Produk Baru'; const category = idxCat >= 0 && row[idxCat] !== undefined ? row[idxCat] : 'Oral'; const stockVal = idxStock >= 0 && row[idxStock] !== undefined ? Number(row[idxStock]) : 0; const unit = idxUnit >= 0 && row[idxUnit] !== undefined ? String(row[idxUnit]).trim() : 'Tablet'; const batchNo = idxBatch >= 0 && row[idxBatch] !== undefined ? String(row[idxBatch]).trim() : 'BATCH-NEW'; const expDate = idxExp >= 0 && row[idxExp] !== undefined ? parseExcelDate(row[idxExp]) : '2027-12-31'; const price = idxPrice >= 0 && row[idxPrice] !== undefined ? Number(row[idxPrice]) : 1000; const minStock = idxMin >= 0 && row[idxMin] !== undefined ? Number(row[idxMin]) : 10; if (id && name) { const existingIdx = newItems.findIndex(item => item.id === String(id).trim()); const itemObj = { id: String(id).trim(), name: String(name).trim(), category: String(category).trim(), initialStock: isNaN(stockVal) ? 0 : stockVal, unit: unit, minStock: isNaN(minStock) ? 10 : minStock, price: isNaN(price) ? 1000 : price, isLasa: false, isHighAlert: false, expDate: expDate, batchNo: batchNo }; if (existingIdx >= 0) { newItems[existingIdx] = { ...newItems[existingIdx], ...itemObj }; } else { newItems.push(itemObj); } importedCount++; } } setItems(newItems); showAlert(`Berhasil mengimpor ${importedCount} produk dari file Excel! Tanggal & satuan otomatis disesuaikan.`); } catch (err) { console.error(err); showAlert('Gagal membaca file Excel.', 'error'); } }; reader.readAsArrayBuffer(file); };
